   423→  const runKnowledgeAction = (action: string, opts: { id?: string; version?: number; reason?: string }, needsConfirm = false) => {

   430→      }

   432→      setKnowledgeConfirm(null);
   433→    }
   434→    setKnowledgeActionError(null);
   435→    knowledgeAction(action, opts, (r) => {
   436→      if (r?.ok === false) setKnowledgeActionError(r?.error || `knowledge: ${action} failed`);

   439→      knowledgeAction('refresh', {}, () => setKnowledgeRefreshing(false));
   440→      setTimeout(() => setKnowledgeRefreshing(false), 30000);
   441→    });
   442→  };

   450→    if (!paired) return;

   462→        setVBusy(false);
   463→        if (r?.ok && r.detail) setVRun(r.detail);
   464→        else if (r?.ok === false) setVError(r.error || 'could not load the run');
   465→      });
   466→    }
   467→  };

   469→  const runVAction = (kind: 'start' | 'cancel', runId: string, dangerous = false) => {

   471→      setVConfirm(`vstart:${runId}`);
   472→      if (vConfirmTimer.current) clearTimeout(vConfirmTimer.current);
   473→      vConfirmTimer.current = setTimeout(() => setVConfirm(null), 5000);
   474→      return;
   475→    }
   476→    if (vConfirmTimer.current) clearTimeout(vConfirmTimer.current);
   477→    setVConfirm(null);
   478→    setVError(null);
   479→    setVBusy(true);
   480→    verificationAction(kind, { runId, confirmDangerous: dangerous }, (r) => {
   481→      setVBusy(false);
   482→      if (r?.ok === false) setVError(r.error || `verification: ${kind} failed`);
   483→    });
   484→  };

   486→  const sendVFeedback = (runId: string, decision: 'accept' | 'reject' | 'comment', comment: string) => {

   489→      setVBusy(false);
   490→      if (r?.ok === false) setVError(r.error || 'feedback failed');
   491→    });
   492→  };
   493→
   494→  const wsOnline = !!status.workspaceOnline;
   495→
   496→  /* ---------------------------------------------------------------- */
   497→  /* Derived data                                                      */
   498→  /* ---------------------------------------------------------------- */
   499→  const stats = useMemo(() => {
   500→    let reads = 0, writes = 0, edits = 0, deletes = 0, notes = 0, localEvents = 0;
   501→    for (const e of entries) {
   502→      if (e.kind === 'tool' && e.source !== 'system') {
   503→        if (e.tool === 'read_file') reads++;
   504→        else if (e.tool === 'write_file') writes++;
   505→        else if (e.tool === 'edit_file') edits++;
   506→        else if (e.tool === 'delete_file') deletes++;
   507→      }
   508→      if (e.kind === 'note') notes++;
   509→      if (e.kind === 'fs') localEvents++;
   510→    }
   511→    return { reads, writes, edits, deletes, notes, localEvents };
   512→  }, [entries]);
   513→
   514→  const feed = useMemo(() => {
   515→    const list = entries.filter((e) =>
   516→      feedFilter === 'all' ? true
   517→        : feedFilter === 'ai' ? (e.kind === 'tool' && e.source !== 'system') || e.kind === 'note' || e.kind === 'skill' || e.kind === 'knowledge'
   518→          : feedFilter === 'you' ? e.source === 'local'
   519→            : e.source === 'system' || (e.kind === 'conn'));
   520→    return list.slice(0, 150);
   521→  }, [entries, feedFilter]);
   522→
   523→  /* flattened visible rows for the explorer */
   524→  const rows = useMemo(() => {
   525→    if (!tree?.tree?.children) return [] as { node: TreeNode; depth: number }[];
   526→    const out: { node: TreeNode; depth: number }[] = [];
   527→    const q = filter.trim().toLowerCase();
   528→
   529→    if (q) {
   530→      // flat search across the whole tree
   531→      const hits: TreeNode[] = [];
   532→      const walk = (n: TreeNode) => {
   533→        for (const c of n.children ?? []) {
   534→          if (c.type === 'file' && c.path.toLowerCase().includes(q)) hits.push(c);
   535→          if (c.type === 'dir') walk(c);
   536→        }
   537→      };
   538→      walk(tree.tree);
   539→      return hits.slice(0, showAllFlat ? 1000 : 300).map((node) => ({ node, depth: 0 }));
   540→    }
   541→
   542→    const walk = (n: TreeNode, depth: number) => {
   543→      for (const c of n.children ?? []) {
   544→        out.push({ node: c, depth });
   545→        if (c.type === 'dir' && expanded.has(c.path)) walk(c, depth + 1);
   546→      }
   547→    };
   548→    walk(tree.tree, 0);
   549→    return out.slice(0, 2500);
   550→  }, [tree, expanded, filter, showAllFlat]);
   551→
   552→  const totalFlatHits = useMemo(() => {
   553→    if (!filter.trim() || !tree?.tree) return 0;
   554→    let count = 0;
   555→    const q = filter.trim().toLowerCase();
   556→    const walk = (n: TreeNode) => {
   557→      for (const c of n.children ?? []) {
   558→        if (c.type === 'file' && c.path.toLowerCase().includes(q)) count++;
   559→        if (c.type === 'dir') walk(c);
   560→      }
   561→    };
   562→    walk(tree.tree);
   563→    return count;
   564→  }, [tree, filter]);
   565→
   566→  /* knowledge derivations */
   567→  const knowledgeCounts = knowledge?.counts;
   568→  const knowledgeTotal = knowledgeCounts
   569→    ? (knowledgeCounts.verified ?? 0) + (knowledgeCounts.candidate ?? 0) + (knowledgeCounts.verificationPending ?? 0)
   570→      + (knowledgeCounts.rejected ?? 0) + (knowledgeCounts.deprecated ?? 0) + (knowledgeCounts.archived ?? 0)
   571→    : 0;
   572→  const knowledgeEmpty = !!knowledgeCounts
   573→    && (knowledgeCounts.verified ?? 0) === 0 && (knowledgeCounts.candidate ?? 0) === 0 && (knowledgeCounts.verificationPending ?? 0) === 0
   574→    && (knowledgeCounts.rejected ?? 0) === 0 && (knowledgeCounts.deprecated ?? 0) === 0 && (knowledgeCounts.archived ?? 0) === 0;
   575→
   576→  const knowledgeCategories = useMemo(() => {
   577→    const set = new Set<string>();
   578→    for (const r of knowledge?.experiences ?? []) if (r.category) set.add(r.category);
   579→    return Array.from(set).sort();
   580→  }, [knowledge]);
   581→
   582→  const knowledgeRecords = useMemo(() => {
   583→    const q = knowledgeSearch.trim().toLowerCase();
   584→    return (knowledge?.experiences ?? []).filter((r) => {
   585→      if (knowledgeScope !== 'all' && r.scope !== knowledgeScope) return false;
   586→      if (knowledgeStatusFilter !== 'all' && r.status !== knowledgeStatusFilter) return false;
   587→      if (knowledgeCategory && r.category !== knowledgeCategory) return false;
   588→      if (q && !`${r.id} ${r.title} ${r.problemSummary} ${r.category}`.toLowerCase().includes(q)) return false;
   589→      return true;
   590→    });
   591→  }, [knowledge, knowledgeSearch, knowledgeScope, knowledgeStatusFilter, knowledgeCategory]);
   592→
   593→  const selectedKnowledge = useMemo(
   594→    () => (knowledgeSelected ? (knowledge?.experiences ?? []).find((r) => r.id === knowledgeSelected) ?? null : null),
   595→    [knowledge, knowledgeSelected]
   596→  );
   597→
   598→  const selectedKnowledgeVersions = useMemo(
   599→    () => (selectedKnowledge?.versions ?? []).slice().sort((a, b) => b.version - a.version),
   600→    [selectedKnowledge]
   601→  );
   602→
   603→  const activeKnowledgeVersion = useMemo(() => {
   604→    if (!selectedKnowledge) return null;
   605→    return (selectedKnowledge.versions ?? []).find((v) => v.version === selectedKnowledge.activeVersion)
   606→      ?? selectedKnowledgeVersions[0]
   607→      ?? null;
   608→  }, [selectedKnowledge, selectedKnowledgeVersions]);
   609→
   610→  const selectedKnowledgeFeedback = useMemo(
   611→    () => (knowledge?.feedback ?? []).filter((f) => f.experienceId === selectedKnowledge?.id),
   612→    [knowledge, selectedKnowledge]
   613→  );
   614→
   615→  const recordRuns = useMemo(
   616→    () => (verification?.runs ?? []).filter((r) => r.knowledgeId === (selectedKnowledge?.id ?? '\u0000')),
   617→    [verification, selectedKnowledge]
   618→  );
   619→
   620→  const toggleDir = (path: string) => {
   621→    setExpanded((prev) => {
   622→      const next = new Set(prev);
   623→      if (next.has(path)) next.delete(path);
   624→      else next.add(path);
   625→      return next;
   626→    });
   627→  };
   628→
   629→  /* nav badge counts */
   630→  const navCounts: Partial<Record<ViewId, number>> = {
   631→    activity: entries.length,
   632→    knowledge: knowledgeTotal,
   633→    skills: skills.length,
   634→  };
   635→
   636→  /* ---------------------------------------------------------------- */
   637→  /* Render                                                            */
   638→  /* ---------------------------------------------------------------- */
   639→  return (
   640→    <div className="min-h-screen bg-vs-bg text-vs-text">
   641→      <div className="flex">
   642→        {/* Sidebar (desktop) */}
   643→        <aside className="hidden lg:flex w-60 flex-none flex-col sticky top-0 h-screen bg-vs-panel border-r border-vs-border">
   644→          {/* brand */}
   645→          <div className="flex items-center gap-2.5 px-4 h-14 border-b border-vs-border/70 flex-none">
   646→            <div className="size-8 rounded-lg bg-vs-blue/15 border border-vs-blue/40 grid place-items-center flex-none">
   647→              <Zap className="size-4 text-vs-blue-bright" />
   648→            </div>
   649→            <div className="min-w-0">
   650→              <div className="text-[13px] font-semibold text-vs-bright leading-tight truncate">Workspace Bridge</div>
   651→              <div className="text-[10px] text-vs-faint leading-tight">z.ai observatory</div>
   652→            </div>
   653→          </div>
   654→
   655→          {/* nav */}
   656→          <nav className="flex-1 overflow-y-auto px-2.5 py-3 space-y-4">
   657→            {NAV_SECTIONS.map((section) => (
   658→              <div key={section.label}>
   659→                <div className="px-3 pb-1.5 text-[10px] uppercase tracking-wider text-vs-faint">{section.label}</div>
   660→                <div className="space-y-0.5">
   661→                  {section.items.map((item) => (
   662→                    <button
   663→                      key={item.id}
   664→                      suppressHydrationWarning
   665→                      onClick={() => goTo(item.id)}
   666→                      className={cn(
   667→                        'w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-[13px] transition-colors',
   668→                        view === item.id ? 'bg-vs-active text-white font-medium' : 'text-vs-dim hover:bg-vs-hover hover:text-vs-text'
   669→                      )}
   670→                      aria-current={view === item.id ? 'page' : undefined}
   671→                    >
   672→                      <span className="flex-none">{item.icon}</span>
   673→                      <span className="flex-1 text-left truncate">{item.label}</span>
   674→                      {(navCounts[item.id] ?? 0) > 0 && (
   675→                        <span className="text-[10px] font-mono text-vs-faint flex-none">{navCounts[item.id]}</span>
   676→                      )}
   677→                    </button>
   678→                  ))}
   679→                </div>
   680→              </div>
   681→            ))}
   682→          </nav>
   683→
   684→          {/* connection status */}
   685→          <div className="flex-none border-t border-vs-border/70 px-4 py-3.5 space-y-2.5">
   686→            <div className="flex items-center gap-2.5 min-w-0" title={`bridge service ${svcUp ? 'online' : 'offline'}`}>
   687→              <Dot tone={svcUp ? 'green' : 'red'} />
   688→              <span className="text-[11px] text-vs-dim flex-1 truncate">service</span>
   689→              <span className={`text-[11px] flex-none ${svcUp ? 'text-vs-green' : 'text-vs-red'}`}>{svcUp ? 'online' : 'offline'}</span>
   690→            </div>
   691→            <div className="flex items-center gap-2.5 min-w-0" title={wsOnline ? status.workspaceName ?? 'workspace online' : 'workspace offline'}>
   692→              <Dot tone={wsOnline ? 'green' : 'yellow'} />
   693→              <span className="text-[11px] text-vs-dim flex-1 truncate">workspace</span>
   694→              <span className={`text-[11px] flex-none truncate max-w-[7rem] ${wsOnline ? 'text-vs-text' : 'text-vs-yellow'}`}>
   695→                {wsOnline ? status.workspaceName ?? 'online' : 'offline'}
   696→              </span>
   697→            </div>
   698→            <div className="flex items-center gap-2.5 min-w-0">
   699→              <Dot tone={paired ? 'green' : 'dim'} />
   700→              <span className="text-[11px] text-vs-dim flex-1 truncate">browser</span>
   701→              <span className={`text-[11px] flex-none ${paired ? 'text-vs-green' : 'text-vs-faint'}`}>{paired ? 'paired' : 'unpaired'}</span>
   702→            </div>
   703→          </div>
   704→        </aside>
   705→
   706→        {/* Main column */}
   707→        <div className="flex-1 min-w-0 flex flex-col min-h-screen">
   708→          {/* Mobile header */}
   709→          <header className="lg:hidden sticky top-0 z-30 bg-vs-panel/95 backdrop-blur border-b border-vs-border flex-none">
   710→            <div className="flex items-center gap-2.5 px-4 h-12">
   711→              <div className="size-7 rounded-md bg-vs-blue/15 border border-vs-blue/40 grid place-items-center flex-none">
   712→                <Zap className="size-3.5 text-vs-blue-bright" />
   713→              </div>
   714→              <span className="text-[13px] font-semibold text-vs-bright truncate">Workspace Bridge</span>
   715→              <div className="flex-1" />
   716→              <Dot tone={svcUp ? 'green' : 'red'} className="size-1.5" />
   717→              <Dot tone={wsOnline ? 'green' : 'yellow'} className="size-1.5" />
   718→              {paired && (
   719→                <Badge variant="outline" className="border-vs-green/40 text-vs-green text-[10px] px-1.5">paired</Badge>
   720→              )}
   721→            </div>
   722→            <nav className="flex gap-1 px-3 pb-2 overflow-x-auto" aria-label="Views">
   723→              {NAV_SECTIONS.flatMap((s) => s.items).map((item) => (
   724→                <button
   725→                  key={item.id}
   726→                  suppressHydrationWarning
   727→                  onClick={() => goTo(item.id)}
   728→                  className={cn(
   729→                    'flex-none flex items-center gap-1.5 px-3 py-1.5 rounded-md text-xs whitespace-nowrap transition-colors',
   730→                    view === item.id ? 'bg-vs-active text-white font-medium' : 'text-vs-dim hover:bg-vs-hover hover:text-vs-text'
   731→                  )}
   732→                  aria-current={view === item.id ? 'page' : undefined}
   733→                >
   734→                  <span className="flex-none -ml-0.5">{item.icon}</span>
   735→                  {item.label}
   736→                  {(navCounts[item.id] ?? 0) > 0 && (
   737→                    <span className="text-[10px] font-mono opacity-70">{navCounts[item.id]}</span>
   738→                  )}
   739→                </button>
   740→              ))}
   741→            </nav>
   742→          </header>
   743→
   744→          {/* Content */}
   745→          <main className="flex-1 w-full max-w-[1400px] mx-auto px-4 sm:px-6 py-5 min-w-0">
   746→            {view === 'overview' && (
   747→              <OverviewView
   748→                paired={paired}
   749→                svcUp={svcUp}
   750→                wsOnline={wsOnline}
   751→                status={status}
   752→                tree={tree}
   753→                skills={skills}
   754→                knowledge={knowledge}
   755→                verification={verification}
   756→                stats={stats}
   757→                feed={feed}
   758→                onGoTo={goTo}
   759→                pairInput={pairInput}
   760→                onPairInputChange={setPairInput}
   761→                onPair={pair}
   762→                onPasteAndPair={pasteAndPair}
   763→                pairingNote={pairingNote}
   764→              />
   765→            )}
   766→
   767→            {view === 'activity' && (
   768→              <ActivityView
   769→                paired={paired}
   770→                feed={feed}
   771→                feedFilter={feedFilter}
   772→                onFilterChange={setFeedFilter}
   773→                openIds={openIds}
   774→                onToggleEntry={toggleEntry}
   775→                onClear={clearActivity}
   776→                rawEvents={rawEvents}
   777→                onClearRaw={() => setRawEvents([])}
   778→              />
   779→            )}
   780→
   781→            {view === 'workspace' && (
   782→              <WorkspaceView
   783→                paired={paired}
   784→                wsOnline={wsOnline}
   785→                tree={tree}
   786→                treeLoading={treeLoading}
   787→                filter={filter}
   788→                onFilterChange={setFilter}
   789→                onRefresh={refreshTree}
   790→                rows={rows}
   791→                expandedPaths={expanded}
   792→                onToggleDir={toggleDir}
   793→                onOpenFile={readFile}
   794→                fileView={fileView}
   795→                onCloseFile={() => setFileView(null)}
   796→                totalFlatHits={totalFlatHits}
   797→                showAllFlat={showAllFlat}
   798→                onShowAllFlat={setShowAllFlat}
   799→                onCollapseAll={() => setExpanded(new Set(['']))}
   800→                workspaceName={status.workspaceName}
   801→              />
   802→            )}
   803→
   804→            {view === 'knowledge' && (
   805→              <KnowledgeView
   806→                paired={paired}
   807→                knowledge={knowledge}
   808→                refreshing={knowledgeRefreshing}
   809→                onRefresh={refreshKnowledge}
   810→                search={knowledgeSearch}
   811→                onSearchChange={setKnowledgeSearch}
   812→                scope={knowledgeScope}
   813→                onScopeChange={setKnowledgeScope}
   814→                statusFilter={knowledgeStatusFilter}
   815→                onStatusFilterChange={setKnowledgeStatusFilter}
   816→                category={knowledgeCategory}
   817→                onCategoryChange={setKnowledgeCategory}
   818→                records={knowledgeRecords}
   819→                categories={knowledgeCategories}
   820→                counts={knowledgeCounts}
   821→                total={knowledgeTotal}
   822→                empty={knowledgeEmpty}
   823→                selected={selectedKnowledge}
   824→                activeVersion={activeKnowledgeVersion}
   825→                selectedVersions={selectedKnowledgeVersions}
   826→                feedback={selectedKnowledgeFeedback}
   827→                onSelect={selectKnowledge}
   828→                versionsOpen={knowledgeVersionsOpen}
   829→                onVersionsOpenChange={setKnowledgeVersionsOpen}
   830→                openVersions={knowledgeOpenVersions}
   831→                onToggleVersion={toggleKnowledgeVersion}
   832→                attemptsOpen={knowledgeAttemptsOpen}
   833→                onAttemptsOpenChange={setKnowledgeAttemptsOpen}
   834→                auditOpen={knowledgeAuditOpen}
   835→                onAuditOpenChange={setKnowledgeAuditOpen}
   836→                confirm={knowledgeConfirm}
   837→                actionError={knowledgeActionError}
   838→                onAction={runKnowledgeAction}
   839→                recordRuns={recordRuns}
   840→                vRunId={vRunId}
   841→                onSelectVRun={selectVRun}
   842→                vRun={vRun}
   843→                vBusy={vBusy}
   844→                vError={vError}
   845→                vConfirm={vConfirm}
   846→                onVStart={(runId, dangerous) => runVAction('start', runId, dangerous)}
   847→                onVCancel={(runId) => runVAction('cancel', runId)}
   848→                onVFeedback={sendVFeedback}
   849→                onOpenFile={openFileInWorkspace}
   850→                verificationTotal={verification?.counts?.total ?? 0}
   851→              />
   852→            )}
   853→
   854→            {view === 'skills' && (
   855→              <SkillsView
   856→                paired={paired}
   857→                wsOnline={wsOnline}
   858→                skills={skills}
   859→                skillsMeta={skillsMeta}
   860→                refreshing={skillsRefreshing}
   861→                onRefresh={refreshSkills}
   862→                skillView={skillView}
   863→                onClose={() => setSkillView(null)}
   864→                onOpen={openSkill}
   865→                onSelectFile={selectSkillFile}
   866→              />
   867→            )}
   868→
   869→            {view === 'setup' && (
   870→              <SetupView
   871→                paired={paired}
   872→                svcUp={svcUp}
   873→                wsOnline={wsOnline}
   874→                status={status}
   875→                pageUrl={pageUrl}
   876→                pairInput={pairInput}
   877→                onPairInputChange={setPairInput}
   878→                onPair={pair}
   879→                onPasteAndPair={pasteAndPair}
   880→                pairingNote={pairingNote}
   881→              />
   882→            )}
   883→          </main>
   884→
   885→          {/* Status bar — VS Code-style observability strip */}
   886→          <footer className="mt-auto bg-vs-blue text-white text-[10.5px] font-mono min-h-7 px-3 py-1 pb-[max(0.25rem,env(safe-area-inset-bottom))] flex items-center gap-4 overflow-hidden whitespace-nowrap select-none">
   887→            <span className="flex items-center gap-1.5 shrink-0">
   888→              <Zap className="size-3" />
   889→              <span className="hidden sm:inline">workspace bridge</span>
   890→            </span>
   891→            <span className="flex items-center gap-1.5 shrink-0">
   892→              <span className={`size-1.5 rounded-full inline-block ${svcUp ? 'bg-white' : 'bg-white/30'}`} />
   893→              {svcUp ? 'service online' : 'service offline'}
   894→            </span>
   895→            <span className="flex items-center gap-1.5 shrink-0">
   896→              <span className={`size-1.5 rounded-full inline-block ${wsOnline ? 'bg-white' : 'bg-white/30'}`} />
   897→              {wsOnline ? status.workspaceName ?? 'workspace' : 'workspace offline'}
   898→            </span>
   899→            {paired && <span className="shrink-0 hidden sm:inline">paired</span>}
   900→            <span className="flex-1" />
   901→            {tree?.stats && <span className="shrink-0 hidden md:inline">{tree.stats.files.toLocaleString()} files</span>}
   902→            {skills.length > 0 && (
   903→              <span className="shrink-0 hidden md:inline">{skills.length} skills</span>
   904→            )}
   905→            {(knowledge?.counts?.verified ?? 0) > 0 && (
   906→              <span className="shrink-0 hidden md:inline">{knowledge?.counts?.verified} verified</span>
   907→            )}
   908→            <span className="shrink-0 hidden md:inline">{entries.length} events</span>
   909→            {tree?.stats?.scannedAt && <span className="shrink-0 hidden lg:inline">scan {timeStr(new Date(tree.stats.scannedAt).getTime())}</span>}
   910→          </footer>
   911→        </div>
   912→      </div>
   913→    </div>
   914→  );
   915→}
