   111→function extractCode(raw: string): string {
   112→  const strict = raw.toUpperCase().match(/[A-HJ-NP-Z2-9]{6}/);
   113→  if (strict) return strict[0];
   114→  const loose = raw.toUpperCase().match(/[A-Z0-9]{6}/);
   115→  return loose ? loose[0] : '';
   116→}
   117→
   118→function fmtBytes(n: number | null | undefined): string {
   119→  if (n === null || n === undefined) return '';
   120→  if (n < 1024) return `${n} B`;
   121→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`;
   122→  return `${(n / (1024 * 1024)).toFixed(1)} MB`;
   123→}
   124→
   125→function timeStr(ts: number): string {
   126→  try { return new Date(ts).toLocaleTimeString([], { hour12: false }); } catch { return ''; }
   127→}

   132→export default function AgentPage() {

   134→  const [status, setStatus] = useState<SvcStatus>({});
   135→  const [paired, setPaired] = useState(false);
   136→  const [pairInput, setPairInput] = useState('');
   137→  const [pairingNote, setPairingNote] = useState<string | null>(null);
   138→
   139→  const [tree, setTree] = useState<TreePayload | null>(null);
   140→  const [treeLoading, setTreeLoading] = useState(false);
   141→  const [expanded, setExpanded] = useState<Set<string>>(new Set(['']));
   142→  const [filter, setFilter] = useState('');
   143→  const [showAllFlat, setShowAllFlat] = useState(false);
   144→
   145→  const [entries, setEntries] = useState<ActivityEntry[]>([]);
   146→  const [openIds, setOpenIds] = useState<Set<string>>(new Set());
   147→  const [feedFilter, setFeedFilter] = useState<'all' | 'ai' | 'you' | 'sys'>('all');
   148→
   149→  const [fileView, setFileView] = useState<{ path: string; loading: boolean; content?: string; meta?: any; error?: string; ts?: number } | null>(null);
   150→
   151→  const [skills, setSkills] = useState<Skill[]>([]);
   152→  const [skillView, setSkillView] = useState<{ name: string; loading: boolean; content?: string; skill?: Skill; error?: string; files?: SkillFileView[]; activeFile?: string } | null>(null);
   153→  const [skillsMeta, setSkillsMeta] = useState<SkillsMeta>({});
   154→  const [skillsRefreshing, setSkillsRefreshing] = useState(false);
   155→
   156→  const [rawEvents, setRawEvents] = useState<{ name: string; data: string; ts: number }[]>([]);
   157→  const [showRaw, setShowRaw] = useState(false);
   158→
   159→  const socketRef = useRef<Socket | null>(null);
   160→  const rawRef = useRef<HTMLDivElement>(null);
   161→  /* Reconnect kit — this page's own URL is the aiCodeAgent.serverUrl value. */
   162→  const [pageUrl, setPageUrl] = useState('');
   163→  useEffect(() => {
   164→    try {
   165→      const u = new URL(window.location.href);
   166→      setPageUrl(u.origin + (u.pathname !== '/' ? u.pathname.replace(/\/+$/, '') : ''));
   167→    } catch { /* ignore */ }
   168→  }, []);
   169→
   170→  const pairedRef = useRef(false);
   171→
   172→  const pushRaw = useCallback((name: string, data: unknown) => {
   173→    setRawEvents((prev) => {
   174→      const next = [...prev.slice(-140), { name, data: (() => { try { return JSON.stringify(data)?.slice(0, 240) ?? ''; } catch { return ''; } })(), ts: Date.now() }];
   175→      return next;
   176→    });
   177→  }, []);

   179→  const upsertEntry = useCallback((e: ActivityEntry) => {
   180→    setEntries((prev) => {
   181→      const i = prev.findIndex((x) => x.id === e.id);
   182→      if (i >= 0) {
   183→        const next = [...prev];
   184→        next[i] = { ...next[i], ...e };
   185→        return next;
   186→      }
   187→      // dedupe notes: the same note may arrive via 'activity' AND 'operator:note'
   188→      if (e.kind === 'note') {
   189→        const dup = prev.find((x) => x.kind === 'note' && x.text === e.text && Math.abs((x.ts ?? 0) - (e.ts ?? 0)) < 5000);
   190→        if (dup) return prev;
   191→      }
   192→      return [e, ...prev].slice(0, 300);
   193→    });
   194→  }, []);
   195→
   196→  /* ---------------------------------------------------------------- */
   197→  /* Socket wiring                                                     */
   198→  /* ---------------------------------------------------------------- */
   199→  useEffect(() => {
   200→    // Connect to the agent service through the Caddy gateway.
   201→    // NEVER use a port in the URL — always XTransformPort.
   202→    // (?bridgePort=… is a debug/testing override; production default is 3003.)
   203→    const bridgePort = new URLSearchParams(window.location.search).get('bridgePort') ?? '3003';
   204→    const socket = io(`/?XTransformPort=${bridgePort}`, {
   205→      transports: ['websocket', 'polling'],
   206→      forceNew: true,
   207→      reconnection: true,
   208→      reconnectionAttempts: Infinity,
   209→      reconnectionDelay: 1000,
   210→      timeout: 15000,
   211→    });
   212→    socketRef.current = socket;
   213→
   214→    socket.on('connect', () => {
   215→      setSvcUp(true);
   216→      socket.emit('hello', { role: 'web', token: localStorage.getItem(TOKEN_KEY) });
   217→    });
   218→
   219→    // If the bridge service ever dies, ask the Next.js server to revive it
   220→    // (GET /api/bridge respawns the agent-service as a detached process).
   221→    let lastRevive = 0;
   222→    const reviveBridge = () => {
   223→      const now = Date.now();
   224→      if (now - lastRevive < 30_000) return;
   225→      lastRevive = now;
   226→      fetch('/api/bridge').catch(() => undefined);
   227→    };
   228→    socket.on('disconnect', () => { setSvcUp(false); reviveBridge(); });
   229→    socket.on('connect_error', () => { setSvcUp(false); reviveBridge(); });
   230→
   231→    socket.on('status', (s: SvcStatus) => { setStatus(s); pushRaw('status', s); });
   232→
   233→    socket.on('pair:ok', (d: { token: string }) => {
   234→      localStorage.setItem(TOKEN_KEY, d.token);

   236→      setPaired(true);
   237→      setPairingNote(null);
   238→      setTreeLoading(true);
   239→      socket.emit('activity:list');
   240→    });
   241→    socket.on('pair:rejected', (d: { reason: string }) => {
   242→      setPairingNote(d?.reason ?? 'Invalid pairing code.');
   243→    });
   244→
   245→    socket.on('activity:list', (d: { entries: ActivityEntry[] }) => {
   246→      setEntries((d?.entries ?? []).slice().reverse().slice(0, 300));
   247→      setTreeLoading(false);
   248→    });

   250→    socket.on('activity:cleared', () => { setEntries([]); });
   251→
   252→    socket.on('operator:note', (d: { text: string; ts: number }) => {
   253→      // paired clients get notes via the 'activity' event; avoid double render
   254→      if (pairedRef.current) return;
   255→      upsertEntry({ id: `note-${d.ts}`, ts: d.ts, source: 'operator', actor: 'GLM (z.ai web)', kind: 'note', text: d.text });
   256→    });
   257→
   258→    /* live pending tools (filled in by the matching activity entry) */
   259→    socket.on('tool:call', (d: { id: string; name: string; args: Record<string, unknown> }) => {
   260→      upsertEntry({
   261→        id: d.id, ts: Date.now(), source: 'operator', actor: '…', kind: 'tool', tool: d.name,
   262→        path: typeof (d.args as any)?.path === 'string' ? String((d.args as any).path) : '',
   263→        args: d.args, pending: true,
   264→      } as ActivityEntry);
   265→      pushRaw('tool:call', d);
   266→    });
   267→    socket.on('tool:result', (d: { id: string; ok: boolean; result: string }) => {
   268→      pushRaw('tool:result', { id: d.id, ok: d.ok });
   269→    });
   270→
   271→    socket.on('workspace:tree', (t: TreePayload) => {
   272→      setTree(t);
   273→      setTreeLoading(false);
   274→      pushRaw('workspace:tree', { stats: t?.stats });
   275→    });
   276→
   277→    socket.on('file:content', (d: { path: string; ok: boolean; content?: string; meta?: any; error?: string }) => {
   278→      setFileView((prev) => prev && prev.path === d.path
   279→        ? { ...prev, loading: false, content: d.ok ? d.content : undefined, meta: d.meta, error: d.ok ? undefined : d.error, ts: Date.now() }
   280→        : prev);
   281→      pushRaw('file:content', { path: d.path, ok: d.ok });
   282→    });
   283→
   284→    socket.on('session:cleared', () => pushRaw('session:cleared', {}));
   285→
   286→    /* skills — documents in the workspace .skills/ folder, learned before GLM thinks */
   287→    socket.on('skill:list', (d: { skills?: Skill[]; scannedAt?: number; online?: boolean; note?: string }) => {
   288→      setSkills(Array.isArray(d?.skills) ? d.skills : []);
   289→      setSkillsMeta({ scannedAt: d?.scannedAt, online: d?.online, note: d?.note });
   290→      pushRaw('skill:list', { count: d?.skills?.length, online: d?.online });
   291→    });
   292→
   293→    return () => {
   294→      socket.disconnect();
   295→    };
   296→  }, [pushRaw, upsertEntry]);
   297→
   298→  useEffect(() => {
   299→    if (showRaw) rawRef.current?.scrollTo({ top: rawRef.current.scrollHeight });
   300→  }, [rawEvents, showRaw]);
   301→
   302→  /* ---------------------------------------------------------------- */
   303→  /* Actions                                                           */
   304→  /* ---------------------------------------------------------------- */
   305→  const pair = (code: string) => {
   306→    const c = extractCode(code || pairInput);
   307→    if (c) {
   308→      socketRef.current?.emit('pair', { code: c });
   309→      setPairInput('');
   310→    } else {
   311→      setPairingNote('Enter the 6-character code shown in VS Code.');
   312→    }
   313→  };
   314→
   315→  const pasteAndPair = async () => {
   316→    try {
   317→      const raw = await navigator.clipboard.readText();
   318→      const c = extractCode(raw);
   319→      if (c) pair(c);
   320→      else setPairingNote('No pairing code found on the clipboard.');

   322→      setPairingNote('Clipboard unavailable — type the code manually.');
   323→    }
   324→  };

   326→  const refreshTree = () => {
   327→    if (!paired) return;
   328→    setTreeLoading(true);
   329→    socketRef.current?.emit('tree:refresh');
   330→    // safety: stop the spinner if no reply in 30s
   331→    setTimeout(() => setTreeLoading(false), 30000);
   332→  };
   333→
   334→  const readFile = (path: string) => {
   335→    if (!paired) return;
   336→    setFileView({ path, loading: true, ts: Date.now() });
   337→    socketRef.current?.emit('file:read', { path });
   338→  };
   339→
   340→  const clearActivity = () => socketRef.current?.emit('activity:clear');

   343→  /* Skills — read from the workspace .skills/ folder                  */

   347→    setSkillsRefreshing(true);

   349→      setSkillsRefreshing(false);

   353→    // safety: stop the spinner if no reply in 30s
   354→    setTimeout(() => setSkillsRefreshing(false), 30000);

   373→  const wsOnline = !!status.workspaceOnline;

   375→  /* ---------------------------------------------------------------- */
   376→  /* Derived data                                                      */
   377→  /* ---------------------------------------------------------------- */
   378→  const stats = useMemo(() => {

   383→        else if (e.tool === 'write_file') writes++;
   384→        else if (e.tool === 'edit_file') edits++;
   385→        else if (e.tool === 'delete_file') deletes++;
   386→      }
   387→      if (e.kind === 'note') notes++;
   388→      if (e.kind === 'fs') localEvents++;

   397→          : feedFilter === 'you' ? e.source === 'local'
   398→            : e.source === 'system' || (e.kind === 'conn'));

   401→
   402→  /* flattened visible rows for the explorer */

   404→    if (!tree?.tree?.children) return [] as { node: TreeNode; depth: number }[];

   406→    const q = filter.trim().toLowerCase();
   407→
   408→    if (q) {
   409→      // flat search across the whole tree
   410→      const hits: TreeNode[] = [];
   411→      const walk = (n: TreeNode) => {
   412→        for (const c of n.children ?? []) {
   413→          if (c.type === 'file' && c.path.toLowerCase().includes(q)) hits.push(c);
   414→          if (c.type === 'dir') walk(c);
   415→        }
   416→      };
   417→      walk(tree.tree);
   418→      return hits.slice(0, showAllFlat ? 1000 : 300).map((node) => ({ node, depth: 0 }));
   419→    }
   420→
   421→    const walk = (n: TreeNode, depth: number) => {
   422→      for (const c of n.children ?? []) {
   423→        out.push({ node: c, depth });
   424→        if (c.type === 'dir' && expanded.has(c.path)) walk(c, depth + 1);
   425→      }
   426→    };
   427→    walk(tree.tree, 0);
   428→    return out.slice(0, 2500);
   429→  }, [tree, expanded, filter, showAllFlat]);
   430→
   431→  const totalFlatHits = useMemo(() => {
   432→    if (!filter.trim() || !tree?.tree) return 0;
   433→    let count = 0;
   434→    const q = filter.trim().toLowerCase();
   435→    const walk = (n: TreeNode) => {
   436→      for (const c of n.children ?? []) {
   437→        if (c.type === 'file' && c.path.toLowerCase().includes(q)) count++;
   438→        if (c.type === 'dir') walk(c);
   439→      }
   440→    };
   441→    walk(tree.tree);
   442→    return count;
   443→  }, [tree, filter]);
   444→
   445→  const toggleDir = (path: string) => {
   446→    setExpanded((prev) => {
   447→      const next = new Set(prev);
   448→      if (next.has(path)) next.delete(path);
   449→      else next.add(path);
   450→      return next;
   451→    });
   452→  };
   453→
   454→  /* ---------------------------------------------------------------- */
   455→  /* Render                                                            */
   456→  /* ---------------------------------------------------------------- */
   457→  return (
   458→    <div className="min-h-screen flex flex-col bg-vs-bg text-vs-text">
   459→      {/* Header */}
   460→      <header className="border-b border-vs-border bg-vs-panel-2 px-4 py-2.5 flex items-center gap-3 flex-wrap sticky top-0 z-20">
   461→        <div className="flex items-center gap-2">
   462→          <div className="size-8 rounded-md bg-vs-blue/15 border border-vs-blue/40 flex items-center justify-center">
   463→            <Zap className="size-4 text-vs-blue-bright" />
   464→          </div>
   465→          <div>
   466→            <h1 className="text-sm font-semibold leading-tight">Z.ai Workspace Bridge</h1>
   467→            <p className="text-[11px] text-vs-dim leading-tight">
   468→              Chat with GLM on z.ai · watch every action on your workspace here
   469→            </p>
   470→          </div>
   471→        </div>
   472→        <div className="flex-1" />
   473→        <div className="flex items-center gap-2 flex-wrap">
   474→          <Badge variant="outline" className={svcUp ? 'border-vs-green/40 text-vs-green' : 'border-vs-border-2 text-vs-dim'}>
   475→            {svcUp ? <Wifi className="size-3 mr-1" /> : <WifiOff className="size-3 mr-1" />}
   476→            {svcUp ? 'server online' : 'server offline'}
   477→          </Badge>
   478→          <Badge variant="outline" className={wsOnline ? 'border-vs-blue-bright/40 text-vs-blue-bright' : 'border-vs-yellow/40 text-vs-yellow'}>
   479→            <TerminalSquare className="size-3 mr-1" />
   480→            {wsOnline ? status.workspaceName ?? 'workspace online' : 'workspace offline'}
   481→          </Badge>
   482→          {paired && (
   483→            <Badge variant="outline" className="border-vs-green/40 text-vs-green">
   484→              <Link2 className="size-3 mr-1" /> paired
   485→            </Badge>
   486→          )}
   487→          {status.tree?.files ? (
   488→            <Badge variant="outline" className="border-vs-border-2 text-vs-dim">
   489→              <HardDrive className="size-3 mr-1" /> {status.tree.files.toLocaleString()} files
   490→            </Badge>
   491→          ) : null}
   492→        </div>
   493→      </header>
   494→
   495→      {/* Body */}
   496→      <div className="flex-1 p-4 max-w-[1500px] w-full mx-auto">
   497→        {/* Pairing banner */}
   498→        {!paired && (
   499→          <Card className="mb-4 bg-vs-panel border-vs-border">
   500→            <CardContent className="px-4 py-4 flex flex-col sm:flex-row sm:items-center gap-4">
   501→              <div className="flex-1 min-w-0">
   502→                <div className="text-xs font-semibold uppercase tracking-wide text-vs-text flex items-center gap-1.5">
