     1→/**
     2→ * knowledge.ts — the Workspace Bridge Adaptive Intelligence layer (v1).
     3→ *
     4→ * A JSONL knowledge store that lives in the USER'S workspace (knowledge/
     5→ * folder) and is synced through the connected VS Code extension. The
     6→ * workspace is the single source of truth — copy the folder to migrate
     7→ * every record, attempt, feedback and audit line to another machine.
     8→ *
     9→ * Lifecycle (plan §5, first-release subset):
    10→ *   candidate → verification_pending → verified | rejected
    11→ *   then: deprecated / archived (owner), deleted (soft)
    12→ *   Failed attempts are separate records used as negative retrieval signal.
    13→ *
    14→ * Core principle (plan §1): GLM may PROPOSE knowledge; the bridge controls
    15→ * verification, promotion and auditability. The operator (admin API) can
    16→ * submit/attempt/feedback/propose/verify — but can never set "verified"
    17→ * directly. Promotion happens ONLY through:
    18→ *   a) the verification gate: checks executed BY THE SERVICE through the
    19→ *      bridge tool layer (real exit codes captured from the extension), or
    20→ *   b) explicit owner approval from the paired dashboard.
    21→ *
    22→ * Store files (workspace-relative):
    23→ *   knowledge/experiences.jsonl   one experience per line (nested versions)
    24→ *   knowledge/attempts.jsonl      unsuccessful attempts (negative signal)
    25→ *   knowledge/feedback.jsonl      reuse feedback (worked / failed / incompatible)
    26→ *   knowledge/audit.jsonl         append-only audit history
    27→ *   knowledge/meta.json           schema version + ID counters
    28→ *   knowledge/README.md           self-documentation (seeded once)
    29→ */

   327→function tokenize(s: string): Set<string> {
   328→  return new Set(
   329→    String(s ?? '')
   330→      .toLowerCase()
   331→      .split(/[^a-z0-9+#.]+/)
   332→      .filter((t) => t.length > 1),
   333→  );
   334→}

   336→/** 0.5 * Jaccard(doc, query) + 0.5 * fraction of query tokens covered. */
   337→function textOverlap(queryTokens: Set<string>, doc: string): number {
   338→  if (queryTokens.size === 0) return 0;
   339→  const docTokens = tokenize(doc);
   340→  if (docTokens.size === 0) return 0;
   341→  let inter = 0;
   342→  for (const t of queryTokens) if (docTokens.has(t)) inter++;
   343→  const union = docTokens.size + queryTokens.size - inter;
   344→  const jaccard = union > 0 ? inter / union : 0;
   345→  const coverage = inter / queryTokens.size;
   346→  return 0.5 * jaccard + 0.5 * coverage;
   347→}

   407→async function wsReadFile(path: string): Promise<{ ok: boolean; text: string; missing: boolean }> {
   408→  // injected lazily to avoid a circular import at module scope
   409→  const r = await rawExec('read_file', { path });
   410→  if (r.ok) {
   411→    let text = typeof (r as any).content === 'string' ? (r as any).content : String(r.result ?? '');
   412→    if (!(r as any).content && text.startsWith(path)) {
   413→      const idx = text.indexOf('):\n\n'); // strip the "<path> (size, N lines):" preamble
   414→      if (idx > 0) text = text.slice(idx + 4);
   415→    }
   416→    if (text.length > MAX_SYNC_READ_CHARS) {
   417→      return { ok: false, text: '', missing: false };
   418→    }
   419→    return { ok: true, text, missing: false };
   420→  }
   421→  const msg = String(r.result ?? '');
   422→  return { ok: false, text: '', missing: /not (exist|found)|no such/i.test(msg) };
   423→}
   424→
   425→async function wsWriteFile(path: string, content: string): Promise<boolean> {

   427→  return r.ok;

   439→function parseJsonl<T>(text: string): { items: T[]; badLines: number } {
   440→  const items: T[] = [];
   441→  let badLines = 0;
   442→  for (const line of String(text ?? '').split('\n')) {
   443→    const l = line.trim();
   444→    if (!l) continue;
   445→    try {
   446→      items.push(JSON.parse(l) as T);
   447→    } catch {
   448→      badLines++;
   449→    }
   450→  }
   451→  return { items, badLines };
   452→}
   453→
   454→function toJsonl(items: unknown[]): string {
   455→  return items.map((i) => JSON.stringify(i)).join('\n') + (items.length ? '\n' : '');
   456→}

   462→function audit(
   463→  actor: string,
   464→  operation: string,
   465→  entityType: AuditRecord['entityType'],
   466→  entityId: string,
   467→  opts: { oldVersion?: number | null; newVersion?: number | null; reason?: string } = {},
   468→): AuditRecord {
   469→  const rec: AuditRecord = {
   470→    id: nextId('aud'),
   471→    ts: Date.now(),
   472→    actor,
   473→    operation,
   474→    entityType,
   475→    entityId,
   476→    oldVersion: opts.oldVersion ?? null,
   477→    newVersion: opts.newVersion ?? null,
   478→    reason: truncate(opts.reason ?? '', 500),
   479→  };
   480→  store.audit.push(rec);
   481→  if (store.audit.length > MAX_AUDIT_IN_MEMORY) {
   482→    store.audit.splice(0, store.audit.length - MAX_AUDIT_IN_MEMORY);
   483→  }
   484→  // Every audit append must persist: audit.jsonl is append-only history, and
   485→  // a line that exists only in memory is lost on restart. (This exact gap
   486→  // dropped 9 propose audit lines on 2026-11-19 — see AUD-000036 in the
   487→  // store. Call sites that also set dirty.audit are harmless duplicates.)
   488→  dirty.audit = true;
   489→  return rec;
   490→}

   503→/* Serialize mutations: one at a time, in order. */
   504→function enqueue<T>(fn: () => Promise<T>): Promise<T> {
   505→  const run = mutationChain.then(fn, fn);
   506→  mutationChain = run.catch(() => undefined);
   507→  return run;
   508→}

   510→/* Dirty-collection tracking so a mutation only writes changed files. */
   511→const dirty = { experiences: false, attempts: false, feedback: false, audit: false, meta: false };
   512→
   513→async function persistDirty(actor: string): Promise<{ ok: boolean; error?: string }> {
   514→  /* No isOnline gate: bun realms can split the liveness flag from the live
   515→   * socket. The writes themselves fail honestly when the workspace is truly
   516→   * offline — that error is the real signal, not a cached flag. */
   517→  if (!deps) {
   518→    return { ok: false, error: 'module not initialized' };
   519→  }
   520→  const writes: Array<Promise<boolean>> = [];
   521→  if (dirty.experiences) writes.push(wsWriteFile(FILES.experiences, toJsonl(store.experiences)));
   522→  if (dirty.attempts) writes.push(wsWriteFile(FILES.attempts, toJsonl(store.attempts)));
   523→  if (dirty.feedback) writes.push(wsWriteFile(FILES.feedback, toJsonl(store.feedback)));
   524→  if (dirty.audit) writes.push(wsWriteFile(FILES.audit, toJsonl(store.audit)));
   525→  if (dirty.meta) writes.push(wsWriteFile(FILES.meta, JSON.stringify(store.meta, null, 2)));
   526→  const results = await Promise.all(writes);
   527→  dirty.experiences = dirty.attempts = dirty.feedback = dirty.audit = dirty.meta = false;
   528→  persistLocalCache();
   529→  if (results.some((ok) => !ok)) {
   530→    return { ok: false, error: 'writing the knowledge store to the workspace failed — retry the operation' };
   531→  }
   532→  void actor;
   533→  return { ok: true };
   534→}
   535→
   536→function markAllDirty() {
   537→  dirty.experiences = dirty.attempts = dirty.feedback = dirty.audit = dirty.meta = true;
   538→}

   568→async function syncFromWorkspace(source: 'operator' | 'web' | 'system', actor: string): Promise<StoreState> {
   569→  if (syncPromise) return syncPromise;
   570→  syncPromise = (async () => {
   571→    /* The WHOLE body runs inside try/finally: the offline early-return used
   572→     * to skip the finally that clears syncPromise, poisoning every future
   573→     * sync in this process with a cached "offline" result (web clients
   574→     * reconnect faster than the extension after a restart — their connect
   575→     * sync hit the offline branch first and the store never re-synced). */
   576→    try {
   577→    /* No isOnline gate: if the workspace is truly offline the meta read below
   578→     * fails and the honest catch reports it. (A previous gate here poisoned
   579→     * syncPromise with a cached "offline" result whenever web clients
   580→     * reconnected faster than the extension after a restart.) */
   581→    try {
   582→      const meta = await wsReadFile(FILES.meta);
   583→      if (meta.missing) {
   584→        if (store.experiences.length > 0 || store.attempts.length > 0) {
   585→          // folder lost on the workspace side but we hold a cache — self-heal
   586→          markAllDirty();
   587→          const w = await persistDirty(actor);
   588→          await wsWriteFile(FILES.readme, STORE_README);
   589→          store.note = w.ok ? 'knowledge folder was missing — cache re-published to the workspace' : 'knowledge folder missing and re-publish failed';
   590→        } else {
   591→          // fresh seed
   592→          store.meta.counters = { exp: 0, att: 0, fbk: 0, aud: 0 };
   593→          markAllDirty();
   594→          await persistDirty(actor);
   595→          await wsWriteFile(FILES.readme, STORE_README);
   596→          store.note = 'knowledge folder created in the workspace — empty store seeded';
   597→        }
   598→      } else if (meta.ok) {
   599→        const parsedMeta = JSON.parse(meta.text) as StoreState['meta'];
   600→        if ((parsedMeta?.schemaVersion ?? 0) > SCHEMA_VERSION) {
   601→          store.online = true;
   602→          store.note = `workspace store schema v${parsedMeta.schemaVersion} is newer than this bridge (v${SCHEMA_VERSION}) — update the bridge before using knowledge`;
   603→          deps?.broadcastState();
   604→          return store;
   605→        }
   606→        const [exp, att, fbk, aud] = await Promise.all([
   607→          wsReadFile(FILES.experiences),
   608→          wsReadFile(FILES.attempts),
   609→          wsReadFile(FILES.feedback),
   610→          wsReadFile(FILES.audit),
   611→        ]);
   612→        const failures: string[] = [];
   613→        const experiences = exp.ok ? parseJsonl<ExperienceRecord>(exp.text) : (failures.push('experiences'), null);
   614→        const attempts = att.ok ? parseJsonl<AttemptRecord>(att.text) : (failures.push('attempts'), null);
   615→        const feedback = fbk.ok ? parseJsonl<FeedbackRecord>(fbk.text) : (failures.push('feedback'), null);
   616→        const auditRecs = aud.ok ? parseJsonl<AuditRecord>(aud.text) : (failures.push('audit'), null);
   617→        store.experiences = experiences ? experiences.items : store.experiences;
   618→        store.attempts = attempts ? attempts.items : store.attempts;
   619→        store.feedback = feedback ? feedback.items : store.feedback;
   620→        store.audit = auditRecs ? auditRecs.items.slice(-MAX_AUDIT_IN_MEMORY) : store.audit;
   621→        store.meta = parsedMeta;
   622→        store.syncedAt = Date.now();
   623→        store.online = true;
   624→        store.note =
   625→          failures.length > 0 ? `some knowledge files could not be read (${failures.join(', ')}) — cached data kept for those collections` : undefined;
   626→        // counters must never go backwards (protect IDs after hand edits)
   627→        for (const e of store.experiences) {
   628→          const n = parseInt(e.id?.slice(4) ?? '0', 10);
   629→          if (Number.isFinite(n) && n > store.meta.counters.exp) store.meta.counters.exp = n;
   630→        }
   631→        for (const a of store.attempts) {
   632→          const n = parseInt(a.id?.slice(4) ?? '0', 10);
   633→          if (Number.isFinite(n) && n > store.meta.counters.att) store.meta.counters.att = n;
   634→        }
   635→        for (const f of store.feedback) {
   636→          const n = parseInt(f.id?.slice(4) ?? '0', 10);
   637→          if (Number.isFinite(n) && n > store.meta.counters.fbk) store.meta.counters.fbk = n;
   638→        }
   639→        for (const a of store.audit) {
   640→          const n = parseInt(a.id?.slice(4) ?? '0', 10);
   641→          if (Number.isFinite(n) && n > store.meta.counters.aud) store.meta.counters.aud = n;
   642→        }
   643→        persistLocalCache();
   644→      } else {
   645→        store.online = true;
   646→        store.note = `could not read ${FILES.meta}: ${truncate(meta.text, 200)}`;
   647→      }
   648→      journal(source, actor, 'knowledge:sync', true,
   649→        `Knowledge store synced with the workspace — ${store.experiences.length} experiences, ${store.attempts.length} attempts, ${store.audit.length} audit lines.`,
   650→        { experiences: store.experiences.length, attempts: store.attempts.length, online: true });
   651→      deps?.broadcastState();
   652→      return store;
   653→    } catch (e: unknown) {
   654→      store.online = false;
   655→      store.note = `sync failed: ${e instanceof Error ? e.message : String(e)}`;
   656→      journal(source, actor, 'knowledge:sync', false, store.note);
   657→      deps?.broadcastState();
   658→      return store;
   659→    }
   660→    } finally {
   661→      syncPromise = null;
   662→    }
   663→  })();
   664→  return syncPromise;
   665→}

   671→function statusWeight(s: ExperienceStatus): number {
   672→  switch (s) {
   673→    case 'verified': return 1.0;
   674→    case 'candidate': return 0.4;
   675→    case 'verification_pending': return 0.3;
   676→    case 'rejected': return 0.15;
   677→    case 'deprecated': return -0.4;
   678→    case 'archived': return -1; // excluded below, kept for completeness
   679→    default: return 0;
   680→  }
   681→}
   682→
   683→function recencyBoost(iso: string): number {
   684→  const t = Date.parse(iso);
   685→  if (!Number.isFinite(t)) return 0;
   686→  const days = Math.max(0, (Date.now() - t) / 86_400_000);
   687→  return 0.15 / (1 + days / 30);
   688→}
   689→
   690→function reuseSuccessRate(r: { used: number; worked: number; failed: number }): number {
   691→  if (r.used <= 0) return 0;
   692→  return (r.worked - r.failed) / r.used; // -1 .. 1
   693→}

   695→export interface SearchOptions {
   696→  scope?: 'global' | 'project' | 'all';
   697→  category?: string;
   698→  limit?: number;
   699→  includeRejected?: boolean;
   700→  preflight?: boolean;
   701→}
   702→
   703→function search(query: string, opts: SearchOptions, source: 'operator' | 'web' | 'system', actor: string): SearchOutcome {
   704→  const q = String(query ?? '').trim();
   705→  if (!q) return { ok: false, query: q, hits: [], attemptWarnings: [], error: 'empty query' };
   706→
   707→  const project = deps?.projectName() ?? null;
   708→  const scopeFilter = opts.scope ?? 'all';
   709→  const limit = Math.max(1, Math.min(20, opts.limit ?? DEFAULT_SEARCH_LIMIT));
   710→  const qTokens = tokenize(q);
   711→
   712→  const hits: SearchHit[] = [];
   713→  for (const exp of store.experiences) {
   714→    if (exp.deletedAt) continue;
   715→    if (exp.status === 'archived') continue;
   716→    if (exp.scope === 'project' && project && exp.projectId && exp.projectId !== project) continue; // project isolation
   717→    if (scopeFilter !== 'all' && exp.scope !== scopeFilter) continue;
   718→    if (opts.category && exp.category !== opts.category) continue;
   719→    if (exp.status === 'rejected' && !opts.includeRejected) continue;
   720→
   721→    const v = activeVersionOf(exp);
   722→    if (!v) continue;
   723→
   724→    const text = [exp.title, exp.problemSummary, exp.category, v.symptoms, v.rootCause, v.solutionSteps].join(' \n ');
   725→    const overlap = textOverlap(qTokens, text);
   726→    if (overlap <= 0) continue; // relevance gate — never surface unrelated records
   727→
   728→    const scopeMatch = exp.scope === 'project' ? (exp.projectId === project ? 1 : 0) : 0.6;
   729→    const score =
   730→      3.0 * overlap +
   731→      0.6 * scopeMatch +
   732→      statusWeight(exp.status) +
   733→      0.2 * reuseSuccessRate(exp.reuseStats) +
   734→      recencyBoost(exp.updatedAt);
   735→
   736→    hits.push({
   737→      id: exp.id,
   738→      title: exp.title,
   739→      scope: exp.scope,
   740→      projectId: exp.projectId,
   741→      category: exp.category,
   742→      status: exp.status,
   743→      activeVersion: exp.activeVersion,
   744→      score: Math.round(score * 1000) / 1000,
   745→      problemSummary: truncate(exp.problemSummary, 400),
   746→      solutionExcerpt: truncate(v.solutionSteps, 600),
   747→      environment: v.environment,
   748→      applicableVersions: v.applicableVersions,
   749→      limitations: v.limitations,
   750→      verificationStatus: v.verificationStatus,
   751→      verificationSummary: v.verificationSummary,
   752→      evidenceCount: v.evidence.length,
   753→      reuseStats: exp.reuseStats,
   754→      updatedAt: exp.updatedAt,
   755→      warning: exp.status === 'deprecated',
   756→    });
   757→  }
   758→
   759→  hits.sort((a, b) => b.score - a.score);
   760→  const top = hits.slice(0, limit);
   761→
   762→  const attemptWarnings = store.attempts
   763→    .filter((a) => !a.experienceId)
   764→    .map((a) => ({
   765→      rec: a,
   766→      score: 3.0 * textOverlap(qTokens, [a.problemSummary, a.approach, a.failureReason].join(' \n ')),
   767→    }))
   768→    .filter((x) => x.score > 0.6)
   769→    .sort((a, b) => b.score - a.score)
   770→    .slice(0, MAX_ATTEMPT_WARNINGS)
   771→    .map((x) => ({
   772→      id: x.rec.id,
   773→      problemSummary: truncate(x.rec.problemSummary, 300),
   774→      approach: truncate(x.rec.approach, 300),
   775→      failureReason: truncate(x.rec.failureReason, 300),
   776→      result: x.rec.result,
   777→      createdAt: x.rec.createdAt,
   778→      score: Math.round(x.score * 1000) / 1000,
   779→    }));
   780→
   781→  const note =
   782→    top.length === 0 && attemptWarnings.length === 0
   783→      ? 'no relevant knowledge found — proceed normally and do not invent prior experience'
   784→      : undefined;
   785→
   786→  journal(source, actor, 'knowledge:' + (opts.preflight ? 'preflight' : 'search'), true,
   787→    `${opts.preflight ? 'Experience preflight' : 'Knowledge search'} “${truncate(q, 120)}” → ${top.length} record(s)${attemptWarnings.length ? `, ${attemptWarnings.length} failed-attempt warning(s)` : ''}${note ? ` (${note})` : ''}.`,
   788→    { query: truncate(q, 200), hits: top.map((h) => h.id), warnings: attemptWarnings.map((w) => w.id), preflight: !!opts.preflight });
   789→
   790→  return { ok: true, query: q, hits: top, attemptWarnings, note };
   791→}

   981→export interface VerifyOutcome {
   982→  ok: boolean;
   983→  id?: string;
   984→  version?: number;
   985→  promoted?: boolean;
   986→  verificationStatus?: VerificationStatus;
   987→  checks?: Array<{ command: string; exitCode: number | null; timedOut: boolean; passed: boolean; journalRef?: string }>;
   988→  error?: string;
   989→}
   990→
   991→async function verifyExperience(
   992→  id: string,
   993→  checks: CheckSpec[],
   994→  source: 'operator' | 'web',
   995→  actor: string,
   996→  versionOverride?: number,
   997→): Promise<VerifyOutcome> {
   998→  return enqueue(async () => {
   999→    const exp = findExperience(id);
  1000→    if (!exp) return { ok: false, error: `unknown experience "${id}"` };
  1001→    const list = (checks ?? []).filter((c) => c && String(c.command ?? '').trim());
  1002→    if (list.length === 0) {
  1003→      return { ok: false, error: 'at least one check command is required — a successful exit code alone is not evidence unless it exercises the original failure' };
  1004→    }
  1005→    if (!deps) {
  1006→      return { ok: false, error: 'knowledge module not initialized' };
  1007→    }
  1008→    const version = exp.versions.find((v) => v.version === (versionOverride ?? exp.activeVersion));
  1009→    if (!version) return { ok: false, error: `version ${versionOverride ?? exp.activeVersion} not found on ${id}` };
  1010→
  1011→    // capture the ORIGINAL state before the pending transition — the
  1012→    // fail/pass branches below must reason about pre-verification trust
  1013→    const originalStatus = exp.status;
  1014→    const originalActive = exp.activeVersion;
  1015→    const wasActiveVerified = originalStatus === 'verified' && originalActive === version.version;
  1016→    const otherTrustedActive = originalStatus === 'verified' && originalActive !== version.version;
  1017→    exp.status = 'verification_pending';
  1018→    exp.updatedAt = nowIso();
  1019→    dirty.experiences = true;
  1020→    await persistDirty(actor);
  1021→
  1022→    const results: VerifyOutcome['checks'] = [];
  1023→    let allPassed = true;
  1024→    let anyUnable = false;
  1025→    for (const c of list) {
  1026→      const r = await deps.runCheck(c.command, c.timeoutMs);
  1027→      const passed = r.ok && !r.timedOut && r.exitCode === 0;
  1028→      if (r.timedOut || r.exitCode === null) anyUnable = true;
  1029→      else if (!passed) allPassed = false;
  1030→      results.push({ command: c.command, exitCode: r.exitCode, timedOut: r.timedOut, passed, journalRef: r.entryId });
  1031→      version.evidence.push({
  1032→        ts: Date.now(),
  1033→        source: 'run_command',
  1034→        command: truncate(c.command, 500),
  1035→        exitCode: r.exitCode,
  1036→        timedOut: r.timedOut,
  1037→        summary: truncate(r.result, 400),
  1038→        journalRef: r.entryId,
  1039→        actor,
  1040→      });
  1041→    }
  1042→    if (version.evidence.length > MAX_EVIDENCE_PER_VERSION) {
  1043→      version.evidence.splice(0, version.evidence.length - MAX_EVIDENCE_PER_VERSION);
  1044→    }
  1045→
  1046→    let promoted = false;
  1047→    let verificationStatus: VerificationStatus;
  1048→    if (anyUnable && allPassed) {
  1049→      verificationStatus = 'unable_to_verify';
  1050→      version.verificationStatus = 'unable_to_verify';
  1051→      version.verificationSummary = `${results.length} check(s) ran but at least one produced no usable exit code (timeout / spawn failure) — NOT promoted`;
  1052→      // honest uncertainty: the verified target stays pending; a proposal that
  1053→      // could not be verified never demotes its trusted active version
  1054→      exp.status = otherTrustedActive ? 'verified' : 'verification_pending';
  1055→    } else if (allPassed) {
  1056→      verificationStatus = 'passed';
  1057→      version.verificationStatus = 'passed';
  1058→      version.verificationSummary = `verification gate PASSED — ${results.length} check(s) executed by the bridge on the workspace, all exit 0 (evidence refs: ${results.map((r) => r.journalRef ?? '?').join(', ')})`;
  1059→      exp.activeVersion = version.version;
  1060→      exp.status = 'verified';
  1061→      promoted = true;
  1062→      audit(actor, 'experience_promote', 'experience', exp.id, {
  1063→        oldVersion: originalActive,
  1064→        newVersion: version.version,
  1065→        reason: 'verification gate: all checks exit 0',
  1066→      });
  1067→    } else {
  1068→      verificationStatus = 'failed';
  1069→      version.verificationStatus = 'failed';
  1070→      version.verificationSummary = `verification gate FAILED — ${results.filter((r) => !r.passed).length}/${results.length} check(s) did not pass`;
  1071→      if (wasActiveVerified) {
  1072→        // the ACTIVE version of a verified record failed re-verification —
  1073→        // honest demotion: no longer recommended (plan §5 state table)
  1074→        exp.status = 'deprecated';
  1075→      } else if (otherTrustedActive) {
  1076→        // a PROPOSAL failed while a different version stays trusted —
  1077→        // the trusted version remains active and verified (plan §16)
  1078→        exp.status = 'verified';
  1079→      } else {
  1080→        exp.status = 'rejected'; // candidate rejected by evidence
  1081→      }
  1082→    }
  1083→    exp.updatedAt = nowIso();
  1084→    dirty.experiences = true;
  1085→    dirty.audit = true;
  1086→    const w = await persistDirty(actor);
  1087→    journal(source, actor, 'knowledge:verify', w.ok && verificationStatus !== 'failed',
  1088→      `${exp.id} v${version.version} verification ${verificationStatus.toUpperCase()}${promoted ? ' → PROMOTED to verified' : ''}: ${results.map((r) => `exit ${r.exitCode ?? 'n/a'}`).join(', ')}.${w.ok ? '' : ` (⚠ store save failed: ${w.error})`}`,
  1089→      { id: exp.id, version: version.version, verificationStatus, promoted, checks: results, saved: w.ok });
  1090→    deps?.broadcastState();
  1091→    return { ok: w.ok, id: exp.id, version: version.version, promoted, verificationStatus, checks: results, error: w.error };
  1092→  });
  1093→}

  1099→export interface ProposalInput {
  1100→  solutionSteps?: string;
  1101→  symptoms?: string;
  1102→  rootCause?: string;
  1103→  environment?: string;
  1104→  applicableVersions?: string;
  1105→  limitations?: string;
  1106→  compatibilityNotes?: string;
  1107→  reason?: string;
  1108→  /**
  1109→   * Optional corrections of RECORD-LEVEL metadata (title / problemSummary).
  1110→   * Used when a technical review retracts a claim that lives outside the
  1111→   * versioned fields. The original values are snapshotted verbatim into the
  1112→   * new version's recordFieldChanges before the record is updated — the
  1113→   * active pointer still does not move until the proposal passes the gate.
  1114→   */
  1115→  title?: string;
  1116→  problemSummary?: string;
  1117→}
  1118→
  1119→async function proposeImprovement(
  1120→  id: string,
  1121→  input: ProposalInput,
  1122→  source: 'operator' | 'web',
  1123→  actor: string,
  1124→): Promise<{ ok: boolean; id?: string; proposedVersion?: number; error?: string }> {
  1125→  return enqueue(async () => {
  1126→    const exp = findExperience(id);
  1127→    if (!exp) return { ok: false, error: `unknown experience "${id}"` };
  1128→    const solution = String(input.solutionSteps ?? '').trim();
  1129→    const reason = String(input.reason ?? '').trim();
  1130→    if (!solution || !reason) return { ok: false, error: 'solutionSteps and reason are required' };
  1131→
  1132→    const secrets = scanSecrets(solution + reason);
  1133→    if (secrets.blocked) {
  1134→      return { ok: false, error: 'proposal blocked: content appears to contain a secret (private key / API token)' };
  1135→    }
  1136→
  1137→    const maxV = exp.versions.reduce((m, v) => Math.max(m, v.version), 0);
  1138→    const proposedVersion = maxV + 1;
  1139→    const previousActive = exp.activeVersion;
  1140→
  1141→    // Record-level corrections (title / problemSummary) are snapshotted BEFORE
  1142→    // the record is touched, so the prior wording survives verbatim on the
  1143→    // version that caused the change (plan §9: revisions, never overwrites).
  1144→    const newTitle = String(input.title ?? '').trim();
  1145→    const newProblem = String(input.problemSummary ?? '').trim();
  1146→    const recordFieldChanges: NonNullable<ExperienceVersion['recordFieldChanges']> = {};
  1147→    if (newTitle && newTitle !== exp.title) {
  1148→      recordFieldChanges.title = { from: exp.title, to: truncate(newTitle, 300) };
  1149→    }
  1150→    if (newProblem && newProblem !== exp.problemSummary) {
  1151→      recordFieldChanges.problemSummary = { from: exp.problemSummary, to: truncate(newProblem, 2000) };
  1152→    }
  1153→    const revisedRecordFields = Object.keys(recordFieldChanges);
  1154→
  1155→    exp.versions.push({
  1156→      version: proposedVersion,
  1157→      createdBy: actor,
  1158→      createdAt: nowIso(),
  1159→      symptoms: truncate(input.symptoms ?? activeVersionOf(exp)?.symptoms ?? '', 2000),
  1160→      rootCause: truncate(input.rootCause ?? activeVersionOf(exp)?.rootCause ?? '', 2000),
  1161→      solutionSteps: truncate(solution, 6000),
  1162→      environment: truncate(input.environment ?? activeVersionOf(exp)?.environment ?? '', 1000),
  1163→      applicableVersions: truncate(input.applicableVersions ?? activeVersionOf(exp)?.applicableVersions ?? '', 500),
  1164→      limitations: truncate(input.limitations ?? '', 1000),
  1165→      compatibilityNotes: truncate(input.compatibilityNotes ?? '', 1000),
  1166→      changeReason: truncate(reason, 1000),
  1167→      verificationStatus: 'pending',
  1168→      verificationSummary: 'proposal awaiting verification — the currently active version stays trusted until this passes the gate',
  1169→      evidence: [],
  1170→      supersedes: previousActive,
  1171→      ...(revisedRecordFields.length > 0 ? { recordFieldChanges } : {}),
  1172→    });
  1173→    // Apply the record-level corrections only after the snapshot is stored.
  1174→    if (recordFieldChanges.title) exp.title = recordFieldChanges.title.to;
  1175→    if (recordFieldChanges.problemSummary) exp.problemSummary = recordFieldChanges.problemSummary.to;
  1176→    // The active pointer does NOT move yet. A previously rejected/deprecated
  1177→    // record becomes a candidate again; a verified one stays verified (v-old).
  1178→    if (exp.status === 'rejected' || exp.status === 'deprecated') exp.status = 'candidate';
  1179→    exp.updatedAt = nowIso();
  1180→    audit(actor, 'experience_propose', 'experience', exp.id, {
  1181→      oldVersion: previousActive,
  1182→      newVersion: proposedVersion,
  1183→      reason: truncate(reason + (revisedRecordFields.length > 0 ? ` [record-level fields revised, originals preserved on v${proposedVersion}: ${revisedRecordFields.join(', ')}]` : ''), 300),
  1184→    });
  1185→    dirty.experiences = true;
  1186→    dirty.meta = true;
  1187→    const w = await persistDirty(actor);
  1188→    journal(source, actor, 'knowledge:propose', w.ok,
  1189→      w.ok
  1190→        ? `Improvement proposal for ${exp.id} stored as version ${proposedVersion} (supersedes v${previousActive}).${revisedRecordFields.length > 0 ? ` Record-level field(s) corrected with originals preserved on v${proposedVersion}: ${revisedRecordFields.join(', ')}.` : ''} The active version stays v${exp.activeVersion} until the proposal passes verification.`
  1191→        : `Proposal for ${exp.id} NOT saved: ${w.error}`,
  1192→      { id: exp.id, proposedVersion, activeVersion: exp.activeVersion, saved: w.ok });
  1193→    deps?.broadcastState();
  1194→    return { ok: w.ok, id: exp.id, proposedVersion, error: w.error };
  1195→  });
  1196→}

  1199→/* Owner actions — paired dashboard ONLY (never the admin API)         */
  1200→/* ------------------------------------------------------------------ */
  1201→
  1202→export type OwnerAction = 'approve' | 'reject' | 'deprecate' | 'archive' | 'restore' | 'delete';
  1203→
  1204→async function ownerAction(
  1205→  id: string,
  1206→  action: OwnerAction,
  1207→  reason: string,
  1208→  actor: string,
  1209→  versionOverride?: number,
  1210→): Promise<{ ok: boolean; record?: ExperienceRecord; error?: string }> {
  1211→  return enqueue(async () => {
  1212→    const exp = findExperience(id);
  1213→    if (!exp) return { ok: false, error: `unknown experience "${id}"` };
  1214→    const why = truncate(reason, 500) || `owner ${action} from dashboard`;
  1215→
  1216→    switch (action) {
  1217→      case 'approve': {
  1218→        const version = exp.versions.find((v) => v.version === (versionOverride ?? exp.activeVersion));
  1219→        if (!version) return { ok: false, error: `version ${versionOverride ?? exp.activeVersion} not found` };
  1220→        const oldActive = exp.activeVersion;
  1221→        version.verificationStatus = 'human_approved';
  1222→        version.verificationSummary = `owner approval on the dashboard by ${actor} — human review stands in for automated checks`;
  1223→        exp.activeVersion = version.version;
  1224→        exp.status = 'verified';
  1225→        audit(actor, 'experience_promote', 'experience', exp.id, { oldVersion: oldActive, newVersion: version.version, reason: why });
  1226→        break;
  1227→      }
  1228→      case 'reject': {
  1229→        const version = exp.versions.find((v) => v.version === (versionOverride ?? exp.activeVersion));
  1230→        if (version && version.verificationStatus === 'pending') {
  1231→          version.verificationStatus = 'failed';
  1232→          version.verificationSummary = `rejected by owner: ${why}`;
  1233→        }
  1234→        exp.status = 'rejected';
  1235→        audit(actor, 'experience_reject', 'experience', exp.id, { reason: why });
  1236→        break;
  1237→      }
  1238→      case 'deprecate': {
  1239→        exp.status = 'deprecated';
  1240→        audit(actor, 'experience_deprecate', 'experience', exp.id, { reason: why });
  1241→        break;
  1242→      }
  1243→      case 'archive': {
  1244→        exp.status = 'archived';
  1245→        exp.archivedAt = nowIso();
  1246→        audit(actor, 'experience_archive', 'experience', exp.id, { reason: why });
  1247→        break;
  1248→      }
  1249→      case 'restore': {
  1250→        if (!exp.archivedAt && exp.status !== 'deprecated' && exp.status !== 'rejected') {
  1251→          return { ok: false, error: 'record is not archived/deprecated/rejected — nothing to restore' };
  1252→        }
  1253→        exp.status = 'candidate'; // honest default: re-verify to re-trust
  1254→        exp.archivedAt = null;
  1255→        audit(actor, 'experience_restore', 'experience', exp.id, { reason: `${why} (restored as candidate — re-verify to re-trust)` });
  1256→        break;
  1257→      }
  1258→      case 'delete': {
  1259→        if (exp.deletedAt) return { ok: false, error: 'record already deleted' };
  1260→        exp.deletedAt = nowIso(); // soft delete — audit history is preserved (plan §9)
  1261→        audit(actor, 'experience_delete', 'experience', exp.id, { reason: why });
  1262→        break;
  1263→      }
  1264→      default:
  1265→        return { ok: false, error: `unknown action "${action}"` };
  1266→    }
  1267→
  1268→    exp.updatedAt = nowIso();
  1269→    dirty.experiences = true;
  1270→    dirty.audit = true;
  1271→    const w = await persistDirty(actor);
  1272→    journal('web', actor, 'knowledge:action', w.ok,
  1273→      w.ok ? `Owner action “${action}” on ${exp.id}: ${why}` : `Owner action “${action}” on ${exp.id} NOT saved: ${w.error}`,
  1274→      { id: exp.id, action, saved: w.ok });
  1275→    deps?.broadcastState();
  1276→    return { ok: w.ok, record: exp, error: w.error };
  1277→  });
  1278→}
